# f11qa-rom-mirror SHA256SUMS
# =============================================================================
#
# Format: <sha256_lowercase_hex>  <relative_path_from_repo_root>
#
# Verification:
#   bash scripts/audit_sha256.sh          # full audit
#   sha256sum -c SHA256SUMS.txt           # standard sha256sum tool (requires cd)
#
# Generation (after vendor):
#   find . -type f \( -name "*.nes" -o -name "*.bin" -o -name "*.log" \) \
#       -not -path "./.git/*" \
#       -not -path "./docs/*" \
#       -not -path "./scripts/*" \
#       | sort \
#       | xargs sha256sum > SHA256SUMS.txt
#
# Maintenance:
#   - vendor new ROM → append new lines
#   - remove ROM      → delete corresponding lines
#   - replace ROM     → verify upstream SHA matches, then replace line
#   - any mismatch    → CI fails → maintainer investigates upstream change
#
# =============================================================================
